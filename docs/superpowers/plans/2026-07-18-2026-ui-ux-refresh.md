**Goal:** Layer 2026 UI/UX trends (floating glass nav, custom cursor, film grain, kinetic typography, scroll parallax, magnetic buttons, bento grids, glass cards, cursor spotlight, 3D tilt cards) onto the existing dark cyber-security portfolio in `portfolio.html`, without changing its visual identity, content, or dependency footprint.

**Architecture:** Single-file HTML/CSS/vanilla-JS site. All new interactive behavior lives in **one new IIFE `<script>` block** appended at the end of the file; each task inserts its code into that same closure so later tasks can reuse two shared flags (`reducedMotion`, `finePointer`) computed once. All new CSS is appended immediately before the existing `</style>` tag. No build step, no test runner — verification is manual browser QA.

**Tech Stack:** Plain HTML/CSS/JS (ES5-style, no modules for new code), existing Three.js r128 (hero background, untouched) and Motion (scroll reveal, untouched) stay as-is. No new dependencies.

  // --- hero: kinetic decode headline ---
  if(!reducedMotion){
    var heroName = document.getElementById('heroName');
    if(heroName){
      var scrambleChars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789!@#$%_/\\';
      var walker = document.createTreeWalker(heroName, NodeFilter.SHOW_TEXT, null);
      var textNodes = [];
      var tn;
      while(tn = walker.nextNode()) textNodes.push(tn);
      textNodes.forEach(function(node){
        var final = node.textContent;
        var len = final.length;
        if(len === 0) return;
        var duration = 700;
        var start = performance.now();
        (function tick(now){
          var p = Math.min((now-start)/duration, 1);
          var revealCount = Math.floor(p*len);
          var out = '';
          for(var i=0;i<len;i++){
            out += (i < revealCount || final[i] === ' ') ? final[i] : scrambleChars[Math.floor(Math.random()*scrambleChars.length)];
          }
          node.textContent = out;
          if(p < 1) requestAnimationFrame(tick);
          else node.textContent = final;
        })(performance.now());
      });
    }
  }

  /* APPEND_POINT */
})();
